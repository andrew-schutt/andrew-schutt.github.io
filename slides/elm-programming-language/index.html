Declarative



    var numbers = [1, 2, 3, 4, 5];

    function doubleNumbers (numbers) {
        const doubled = [];
        const l = numbers.length;
    
        for (let i = 0; i < l; i++) {
            doubled.push(numbers[i] * 2);
        };
    
        return doubled;
    };
